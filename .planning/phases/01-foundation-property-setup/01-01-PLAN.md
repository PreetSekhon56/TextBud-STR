6. Create src/db/connection.ts:
   ```typescript
   import Database from 'better-sqlite3';
   import path from 'path';
   import fs from 'fs';

   let db: Database.Database;

   export function getDb(): Database.Database {
     if (!db) {
       const dbPath = process.env.DATABASE_PATH || './data/textbud.db';
       const dir = path.dirname(dbPath);
       if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
       db = new Database(dbPath);
       db.pragma('journal_mode = WAL');
       db.pragma('foreign_keys = ON');
     }
     return db;
   }

   export function closeDb(): void {
     if (db) db.close();
   }
   ```

7. Create src/db/schema.ts:

   import { getDb } from './connection.js';
   import fs from 'fs';
   import path from 'path';

   export function initDb(): void {
     const db = getDb();
     const migrationPath = path.join(import.meta.dirname, 'migrations', '001_initial.sql');
     const sql = fs.readFileSync(migrationPath, 'utf-8');
     db.exec(sql);
   }

8. Create scripts/smoke-test-db.ts -- a standalone script that verifies DB initialization at runtime:
   ```typescript
   /**
    * DB Smoke Test: Verifies initDb() creates all expected tables.
    * Run with: npx tsx scripts/smoke-test-db.ts
    * Uses a temporary in-memory or temp-file DB so it doesn't touch production data.
    */
   import Database from 'better-sqlite3';
   import fs from 'fs';
   import path from 'path';

   // Use a temp DB path to avoid touching real data
   const tmpDir = path.join(import.meta.dirname, '..', 'data', '.test');
   if (!fs.existsSync(tmpDir)) fs.mkdirSync(tmpDir, { recursive: true });
   const tmpDbPath = path.join(tmpDir, 'smoke-test.db');

   // Clean up any previous test DB
   if (fs.existsSync(tmpDbPath)) fs.unlinkSync(tmpDbPath);

   // Set env so connection.ts uses our temp DB
   process.env.DATABASE_PATH = tmpDbPath;

   // Now import and run initDb
   const { initDb } = await import('../src/db/schema.js');
   const { getDb, closeDb } = await import('../src/db/connection.js');

   initDb();
   const db = getDb();

   // Query sqlite_master for all tables
   const tables = db.prepare(
     "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name"
   ).all() as { name: string }[];

   const tableNames = tables.map(t => t.name);

   const expectedTables = [
     'checklist_items',
     'checklists',
     'cleaners',
     'faq_entries',
     'owners',
     'properties',
     'rooms',
     'setup_sessions',
     'staging_photos',
     'telegram_groups',
   ];

   const missing = expectedTables.filter(t => !tableNames.includes(t));
   const extra = tableNames.filter(t => !expectedTables.includes(t));

   if (missing.length > 0) {
     console.error('FAIL: Missing tables:', missing.join(', '));
     closeDb();
     fs.unlinkSync(tmpDbPath);
     process.exit(1);
   }

   console.log(`PASS: All ${expectedTables.length} tables created successfully.`);
   if (extra.length > 0) {
     console.log(`Note: Extra tables found (OK): ${extra.join(', ')}`);
   }

   // Verify foreign keys are enabled
   const fkStatus = db.pragma('foreign_keys') as { foreign_keys: number }[];
   if (!fkStatus[0] || (fkStatus[0] as any).foreign_keys !== 1) {
     console.error('FAIL: foreign_keys pragma is not enabled');
     closeDb();
     fs.unlinkSync(tmpDbPath);
     process.exit(1);
   }
   console.log('PASS: foreign_keys pragma is ON.');

   // Cleanup
   closeDb();
   fs.unlinkSync(tmpDbPath);
   fs.rmdirSync(tmpDir, { recursive: false });
   console.log('DB smoke test complete.');
   ```

1. Create src/bot.ts:
   ```typescript
   import { Telegraf, Context } from 'telegraf';
   import { getDb } from './db/connection.js';

   export const bot = new Telegraf(process.env.BOT_TOKEN!);

   // /start in DM: register as owner
   bot.start(async (ctx) => {
     if (ctx.chat.type !== 'private') return; // ignore /start in groups

     const db = getDb();
     const telegramId = ctx.from.id;
     const username = ctx.from.username || null;
     const firstName = ctx.from.first_name;

     // Upsert owner
     db.prepare(`
       INSERT INTO owners (telegram_id, username, first_name)
       VALUES (?, ?, ?)
       ON CONFLICT(telegram_id) DO UPDATE SET
         username = excluded.username,
         first_name = excluded.first_name
     `).run(telegramId, username, firstName);

     const owner = db.prepare('SELECT * FROM owners WHERE telegram_id = ?').get(telegramId);

     await ctx.reply(
       `Welcome to TextBud, ${firstName}! I'm your cleaning operations assistant.\n\n` +
       `I help you manage your properties and cleaning teams right here in Telegram.\n\n` +
       `To get started, send "new property" to set up your first property.`
     );
   });

   // Helper: check if a message is from a DM (private chat)
   export function isDM(ctx: Context): boolean {
     return ctx.chat?.type === 'private';
   }

   // Helper: get owner by telegram_id, returns null if not registered
   export function getOwnerByTelegramId(telegramId: number) {
     const db = getDb();
     return db.prepare('SELECT * FROM owners WHERE telegram_id = ?').get(telegramId) || null;
   }
   ```
