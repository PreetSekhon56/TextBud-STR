**buildChecklistKeyboard:**
```typescript
import { Markup } from 'telegraf';

function buildChecklistKeyboard(sessionId: number, roomId: number, items: ChecklistItem[], completedIds: Set<number>) {
  const buttons = items.map(item => {
    const prefix = completedIds.has(item.id) ? '[done] ' : '';
    return [Markup.button.callback(`${prefix}${item.description}`, `c_${sessionId}_${item.id}`)];
  });
  return Markup.inlineKeyboard(buttons);
}
```
